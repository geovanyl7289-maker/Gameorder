if (args.Length == 2 && args[0] == "--file-status")
{
    // Header-only read: no configuration, password attempt, JSON deserialization, or migration.
    var data = EncryptedLocalFile.ReadIfExists(Path.GetFullPath(args[1])) ?? throw new IOException("Requested file does not exist.");
    try
    {
        bool password = EncryptedLocalFile.HasPasswordMagic(data), dpapi = EncryptedLocalFile.HasEncryptedMagic(data);
        Console.WriteLine(JsonSerializer.Serialize(new { envelopeRecognized = password || dpapi, passwordProtected = password,
            dpapiProtected = dpapi, version = (password || dpapi) && data.Length >= 12 ? BinaryPrimitives.ReadUInt32LittleEndian(data.AsSpan(8,4)) : 0,
            bytes = data.Length, ciphertextSha256 = Convert.ToHexString(SHA256.HashData(data)) }));
    }
    finally { CryptographicOperations.ZeroMemory(data); }
    return;
}
if (args.Length == 2 && args[0] == "--live-status")
{
    // Explicit, read-only status mode; never migrates and never serializes account values into the report.
    var encrypted = EncryptedLocalFile.ReadIfExists(Path.GetFullPath(args[1])) ?? throw new IOException("Requested file does not exist.");
    byte[]? plain = null;
    try
    {
        if (EncryptedLocalFile.HasPasswordMagic(encrypted)) throw new InvalidDataException("Password-bound files support header-only --file-status; this mode does not accept passwords.");
        plain = EncryptedLocalFile.Decrypt(encrypted);
        var state = JsonSerializer.Deserialize<LocalState>(plain) ?? throw new InvalidDataException();
        Console.WriteLine(JsonSerializer.Serialize(new { encrypted = true, accounts = state.Accounts.Count, folders = state.Folders.Count,
            records = state.Activity.Count, hiddenAccounts = state.HiddenAccounts.Count, plaintextSha256 = Convert.ToHexString(SHA256.HashData(plain)) }));
    }
    finally { CryptographicOperations.ZeroMemory(encrypted); if (plain is not null) CryptographicOperations.ZeroMemory(plain); }
    return;
}

    return new() { SteamPath="C:\\SyntheticSteam",TaskbarWidgetEnabled=false,TaskbarWidgetDock="left",
        Accounts=[new() { AccountName="synthetic_alpha",DisplayName="演示玩家 🎮",Note=note??Marker,IsFavorite=true,IsManual=true,FolderId=id }],
        Folders=[new() { Id=id,Name="主账号" }],HiddenAccounts=["synthetic_hidden"],
        Activity=[new() { Title="演示记录",Description="Synthetic-History-详情",Kind="success",Category="account",Time=DateTimeOffset.FromUnixTimeSeconds(1700000000) }] };
}
static byte[] Legacy()
{
    var plain = JsonSerializer.Serialize(State());
    return Encoding.UTF8.GetBytes(plain[..^1]+",\"FutureMetadata\":\"preserve-me\"}");
}
static void SameState(LocalState expected,LocalState actual)
{
    Check(expected.SteamPath==actual.SteamPath && expected.TaskbarWidgetEnabled==actual.TaskbarWidgetEnabled && expected.TaskbarWidgetDock==actual.TaskbarWidgetDock,"preferences lost");
    Check(expected.Accounts.Count==actual.Accounts.Count && expected.Accounts[0].Note==actual.Accounts[0].Note && expected.Accounts[0].DisplayName==actual.Accounts[0].DisplayName
        && actual.Accounts[0].IsManual && actual.Accounts[0].IsFavorite && expected.Accounts[0].FolderId==actual.Accounts[0].FolderId,"metadata lost");
    Check(actual.Folders[0].Name==expected.Folders[0].Name && actual.HiddenAccounts.SequenceEqual(expected.HiddenAccounts),"groups/hidden lost");
    Check(actual.Activity[0].Title==expected.Activity[0].Title && actual.Activity[0].Description==expected.Activity[0].Description
        && actual.Activity[0].Time==expected.Activity[0].Time && actual.Activity[0].Kind==expected.Activity[0].Kind && actual.Activity[0].Category==expected.Activity[0].Category,"activity lost");
}
static void CipherOnly(Fixture f)
{
    foreach(var file in Directory.EnumerateFiles(f.Root))
    { var bytes=File.ReadAllBytes(file); Check(EncryptedLocalFile.HasEncryptedMagic(bytes),"unexpected unencrypted file"); Check(!Contains(bytes,Marker)&&!Contains(bytes,"synthetic_alpha")&&!Contains(bytes,"Synthetic-History"),"plaintext found on disk"); }
}
static bool Contains(byte[] bytes,string value) => bytes.AsSpan().IndexOf(Encoding.UTF8.GetBytes(value))>=0 || bytes.AsSpan().IndexOf(Encoding.Unicode.GetBytes(value))>=0;
static void FailsClosed(Fixture f,byte[] original)
{
    var store=new LocalStore(false,f.Path); Check(store.Load().Accounts.Count==0 && store.LoadError is not null && !store.CanSave,"failure not reported");
    Throws<IOException>(()=>store.Save(State())); Check(original.SequenceEqual(File.ReadAllBytes(f.Path)),"original overwritten");
    Check(!Directory.EnumerateFiles(f.Root,"*.tmp").Any(),"failure left temp");
}
static void Damaged(Action<byte[]> mutate)
{ using var f=new Fixture(); new LocalStore(false,f.Path).Save(State()); var bytes=File.ReadAllBytes(f.Path); mutate(bytes); File.WriteAllBytes(f.Path,bytes); FailsClosed(f,bytes); }
static void InvalidLegacy(string json)
{ using var f=new Fixture(); var bytes=Encoding.UTF8.GetBytes(json); File.WriteAllBytes(f.Path,bytes); FailsClosed(f,bytes); }
static void EncodingMigration(Encoding encoding)
{
    using var f=new Fixture(); var text=Encoding.UTF8.GetString(Legacy()); var bytes=encoding.GetPreamble().Concat(encoding.GetBytes(text)).ToArray(); File.WriteAllBytes(f.Path,bytes);
    var store=new LocalStore(false,f.Path); SameState(State(),store.Load()); Check(store.WasMigrated && store.IsEncrypted,"encoding migration failed"); CipherOnly(f);
}
static void Check(bool condition,string message) { if(!condition) throw new Exception(message); }
static void Throws<T>(Action action) where T:Exception
{ try { action(); } catch(T) { return; } throw new Exception("Expected "+typeof(T).Name); }

sealed class Fixture : IDisposable
{
    private static readonly string FixtureParent=System.IO.Path.GetFullPath(System.IO.Path.Combine(AppContext.BaseDirectory,"fixtures"));
    public string Root { get; }=System.IO.Path.Combine(FixtureParent,Guid.NewGuid().ToString("N"));
    public string Path => System.IO.Path.Combine(Root,"accounts.json");
    public Fixture()=>Directory.CreateDirectory(Root);
    public void Dispose()
    {
        var resolved=System.IO.Path.GetFullPath(Root);
        if(!resolved.StartsWith(FixtureParent+System.IO.Path.DirectorySeparatorChar,StringComparison.OrdinalIgnoreCase)
            || !Guid.TryParseExact(System.IO.Path.GetFileName(resolved),"N",out _)) throw new IOException("Unsafe fixture cleanup path");
        Directory.Delete(resolved,true);
    }
}
