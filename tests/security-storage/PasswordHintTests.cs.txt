using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using Gameorder;

static class PasswordHintTests
{
    private const string Password="Synthetic-Hint-Password-Original";
    private const string NewPassword="Synthetic-Hint-Password-Changed";
    private const string Note="SYNTHETIC-HINT-PRIVATE-NOTE-中文";
    public static IReadOnlyList<(string Name,Action Run)> Tests =>
    [
        ("hint is trimmed, DPAPI-protected and visible before password authentication",()=>
        {
            using var f=new Fixture();var service=f.Seed("  童年公园的那件事  ");var cfg=ReadConfig(f.Security);try {using var json=JsonDocument.Parse(cfg);Check(json.RootElement.GetProperty("PasswordHint").GetString()=="童年公园的那件事","hint not normalized");}finally {Clear(cfg);}
            Check(File.ReadAllBytes(f.Security).AsSpan().IndexOf(Encoding.UTF8.GetBytes("童年公园的那件事"))<0,"hint plaintext on disk");
            var cold=f.Service();var store=f.Store(cold);Check(cold.PasswordHint=="童年公园的那件事"&&!cold.IsAuthenticated,"cold hint missing or authenticated");Check(store.Load().Accounts.Count==0&&store.RequiresUnlock,"hint exposed private metadata");
            var hash=Hash(f.Security);Check(!cold.Verify("wrong",out _)&&cold.PasswordHint=="童年公园的那件事"&&hash==Hash(f.Security),"wrong password altered hint");
        }),
        ("hint-only update and clear preserve metadata ciphertext and all password wrapping fields",()=>
        {
            using var f=new Fixture();var service=f.Seed("旧线索");var accountHash=Hash(f.Accounts);var before=ReadConfig(f.Security);
            Check(service.ChangePasswordHint(Password,"  新线索  ",out var error),error);Check(service.PasswordHint=="新线索"&&Hash(f.Accounts)==accountHash,"hint update rewrote metadata");var after=ReadConfig(f.Security);
            try {using var old=JsonDocument.Parse(before);using var current=JsonDocument.Parse(after);foreach(var name in new[]{"Salt","Iterations","Verifier","KeyId","KeyNonce","KeyTag","WrappedKey"}) Check(old.RootElement.GetProperty(name).GetRawText()==current.RootElement.GetProperty(name).GetRawText(),"hint changed cryptographic field");}finally {Clear(before);Clear(after);}
            Check(service.ChangePasswordHint(Password,"",out _)&&service.PasswordHint==""&&Hash(f.Accounts)==accountHash,"hint clear changed data");Check(f.Service().PasswordHint=="","clear not persisted");
        }),
        ("changing password preserves an omitted hint and clears an explicit empty hint",()=>
        {
            using var f=new Fixture();var service=f.Seed("毕业的那一天");var accountHash=Hash(f.Accounts);
            Check(service.ChangePassword(Password,NewPassword,out _)&&service.PasswordHint=="毕业的那一天"&&Hash(f.Accounts)==accountHash,"omitted hint not preserved");
            var cold=f.Service();_=f.Store(cold);Check(cold.PasswordHint=="毕业的那一天"&&!cold.Verify(Password,out _)&&cold.Verify(NewPassword,out _),"password change hint restart");
            Check(cold.ChangePassword(NewPassword,Password,out _,"")&&f.Service().PasswordHint==""&&Hash(f.Accounts)==accountHash,"explicit empty hint not cleared");
        }),
        ("wrong current password cannot edit a hint or password and leaves both files unchanged",()=>
        {
            using var f=new Fixture();_=f.Seed("常用的个人线索");var cold=f.Service();_=f.Store(cold);var cfgHash=Hash(f.Security);var accountHash=Hash(f.Accounts);
            Check(!cold.ChangePasswordHint("incorrect","攻击者的线索",out _)&&!cold.ChangePassword("incorrect",NewPassword,out _,"另一线索"),"wrong password accepted");
            Check(cold.PasswordHint=="常用的个人线索"&&!cold.IsAuthenticated&&cfgHash==Hash(f.Security)&&accountHash==Hash(f.Accounts),"wrong password mutated storage");
        }),
        ("failed hint configuration write preserves the old hint password and metadata",()=>
        {
            using var f=new Fixture();var faults=new FaultStorage(f.Security);var service=f.Seed("原来的线索",faults);var cfgHash=Hash(f.Security);var accountHash=Hash(f.Accounts);faults.FailAt=faults.Writes+1;
            Check(!service.ChangePasswordHint(Password,"未提交的线索",out _)&&service.PasswordHint=="原来的线索"&&cfgHash==Hash(f.Security)&&accountHash==Hash(f.Accounts),"failed hint write changed storage");
            var cold=f.Service();var store=f.Store(cold);Check(cold.PasswordHint=="原来的线索"&&cold.Verify(Password,out _)&&store.Load().Accounts[0].Note==Note,"failed hint invalidated original password");
        }),
        ("hint length controls multiline text and full password are rejected before enabling",()=>
        {
            using var f=new Fixture();var service=f.Service();var store=f.Store(service);store.Save(State());var original=Hash(f.Accounts);
            foreach(var hint in new[]{new string('界',81),"多行\n文字","制表\t符","nul\0值","行\u2028分隔","段\u2029分隔","我记得"+Password+"的内容"})
                Check(!service.Enable(Password,out var error,hint)&&error.Length>0&&!service.Enabled&&!File.Exists(f.Security)&&Hash(f.Accounts)==original,"invalid hint accepted or mutated metadata");
            Check(service.Enable(Password,out _,"  "+new string('界',80)+"  ")&&service.PasswordHint.Length==80,"trimmed 80-character hint rejected");
        }),
        ("password replacement rejects hints containing either full current or new password",()=>
        {
            using var f=new Fixture();var service=f.Seed("原来那句线索");var cfg=Hash(f.Security);var accounts=Hash(f.Accounts);
            foreach(var hint in new[]{"完整的"+Password,"完整的"+NewPassword}) Check(!service.ChangePassword(Password,NewPassword,out var error,hint)&&error.Contains("完整密码"),"full password in hint accepted");
            Check(cfg==Hash(f.Security)&&accounts==Hash(f.Accounts)&&service.PasswordHint=="原来那句线索","invalid replacement mutated storage");
            Check(service.ChangePasswordHint(Password,"生日口令12345678",out _),"setup preserved clue");cfg=Hash(f.Security);
            Check(!service.ChangePassword(Password,"12345678",out var retainedError)&&retainedError.Contains("完整密码")&&cfg==Hash(f.Security),"preserved hint disclosed new password");
        }),
        ("missing hint field in a valid v2 configuration defaults to empty on restart",()=>
        {
            using var f=new Fixture();_=f.Seed("移除这个可选字段");var accounts=Hash(f.Accounts);ReplaceHintField(f.Security,null,remove:true);
            var cold=f.Service();var store=f.Store(cold);Check(cold.PasswordHint==""&&!cold.ConfigurationError&&cold.Verify(Password,out _)&&store.Load().Accounts[0].Note==Note&&Hash(f.Accounts)==accounts,"missing optional hint broke unlock");
        }),
        ("legacy hint-only editing preserves DPAPI accounts and normal unlock migration retains the clue",()=>
        {
            using var f=new Fixture();new LocalStore(false,f.Accounts).Save(State());WriteLegacy(f.Security,"最初那条线索");var original=Hash(f.Accounts);var cold=f.Service();_=f.Store(cold);
            Check(cold.PasswordHint=="最初那条线索"&&!cold.IsAuthenticated&&cold.ChangePasswordHint(Password,"更新的旧版线索",out _)&&!cold.IsAuthenticated&&Hash(f.Accounts)==original,"legacy hint editor migrated metadata");
            var cfg=ReadConfig(f.Security);try{using var json=JsonDocument.Parse(cfg);Check(json.RootElement.GetProperty("Version").GetInt32()==1,"hint-only changed legacy protection version");}finally{Clear(cfg);}
            var migrated=f.Service();var store=f.Store(migrated);Check(migrated.Verify(Password,out _)&&migrated.PasswordHint=="更新的旧版线索"&&EncryptedLocalFile.HasPasswordMagic(File.ReadAllBytes(f.Accounts))&&store.Load().Accounts[0].Note==Note,"normal legacy migration lost hint or data");
        }),
        ("disabling and destructive reset remove the hint while keeping reset isolated",()=>
        {
            using(var f=new Fixture()) {var service=f.Seed("需要被移除的线索");Check(service.Disable(Password,out _)&&!service.Enabled&&service.PasswordHint==""&&f.Service().PasswordHint=="","disabled hint retained");var cfg=ReadConfig(f.Security);try{using var json=JsonDocument.Parse(cfg);Check(json.RootElement.GetProperty("PasswordHint").GetString()=="","disabled config kept hint");}finally{Clear(cfg);}}
            using(var f=new Fixture()) {var context=GameorderDataContext.CreateTest(f.Root);Check(context.LockService.ChangePasswordHint(GameorderDataContext.TestPassword,"仅用于重置的假线索",out _),"reset hint setup");_=context.ResetAsync().GetAwaiter().GetResult();Check(context.LockService.PasswordHint==""&&!File.Exists(context.SecurityPath)&&GameorderDataContext.CreateTest(f.Root).LockService.PasswordHint=="","reset retained hint");}
        }),
        ("malformed stored hint is rejected without exposing it or modifying encrypted files",()=>
        {
            using var f=new Fixture();_=f.Seed("健康线索");var accounts=Hash(f.Accounts);
            foreach(var hint in new[]{new string('x',81),"stored\ncontrol"}) {ReplaceHintField(f.Security,hint);var cfg=Hash(f.Security);var cold=f.Service();_=f.Store(cold);Check(cold.ConfigurationError&&cold.Enabled&&cold.PasswordHint==""&&!cold.Verify(Password,out _)&&cfg==Hash(f.Security)&&accounts==Hash(f.Accounts),"invalid persisted hint did not fail closed");}
        }),
        ("failed disable retains the hint in pending and rolled-back key configuration",()=>
        {
            using var f=new Fixture();var faults=new FaultStorage(f.Security);var service=f.Seed("待禁用仍应保留的线索",faults);var cfgHash=Hash(f.Security);var accountHash=Hash(f.Accounts);faults.FailAt=faults.Writes+2;faults.Observe=data=>{var plain=new AppLockWindowsProtector().Unprotect(data[4..]);try{using var doc=JsonDocument.Parse(plain);if(doc.RootElement.TryGetProperty("PendingDisable",out var pending)&&pending.GetBoolean()) Check(doc.RootElement.GetProperty("PasswordHint").GetString()=="待禁用仍应保留的线索","pending cfg lost hint");}finally{Clear(plain);}};
            Check(!service.Disable(Password,out _)&&service.PasswordHint=="待禁用仍应保留的线索"&&cfgHash==Hash(f.Security)&&accountHash==Hash(f.Accounts),"disable rollback lost hint");Check(f.Service().PasswordHint=="待禁用仍应保留的线索","rollback hint missing after restart");
        })
    ];
    private static LocalState State()=>new(){Accounts=[new(){AccountName="synthetic_hint_user",Note=Note}],Activity=[new(){Title="假提示历史"}]};
    private static void WriteLegacy(string path,string? hint=null)
    {
        var salt=RandomNumberGenerator.GetBytes(32);var verifier=Rfc2898DeriveBytes.Pbkdf2(Password,salt,AppLockService.PasswordIterations,HashAlgorithmName.SHA256,32);
        var plain=JsonSerializer.SerializeToUtf8Bytes(new{Version=1,Enabled=true,Iterations=AppLockService.PasswordIterations,Salt=salt,Verifier=verifier,PasswordHint=hint});
        try{WriteConfig(path,plain);}finally{Clear(plain);Clear(salt);Clear(verifier);}
    }
    private static byte[] ReadConfig(string path)=>new AppLockWindowsProtector().Unprotect(File.ReadAllBytes(path)[4..]);
    private static void WriteConfig(string path,byte[] plain)=>File.WriteAllBytes(path,"GOL1"u8.ToArray().Concat(new AppLockWindowsProtector().Protect(plain)).ToArray());
    private static void ReplaceHintField(string path,string? hint,bool remove=false)
    {
        var plain=ReadConfig(path);byte[]? changed=null;
        try {var values=JsonSerializer.Deserialize<Dictionary<string,JsonElement>>(plain)!;if(remove)values.Remove("PasswordHint");else values["PasswordHint"]=JsonSerializer.SerializeToElement(hint);changed=JsonSerializer.SerializeToUtf8Bytes(values);WriteConfig(path,changed);}
        finally {Clear(plain);Clear(changed);}
    }
    private static string Hash(string path)=>Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(path)));
    private static void Check(bool condition,string message){if(!condition)throw new Exception(message);}
    private static void Clear(byte[]? value){if(value is not null)CryptographicOperations.ZeroMemory(value);}
    private sealed class FaultStorage(string path):IAppLockStorage
    {
        private readonly AppLockFileStorage _inner=new(path);public int Writes;public int FailAt;public Action<byte[]>? Observe;
        public byte[]? Read()=>_inner.Read();public void Delete()=>_inner.Delete();
        public void Write(byte[] data){Observe?.Invoke(data);if(++Writes==FailAt)throw new IOException("Synthetic hint cfg write failure");_inner.Write(data);}
    }
    private sealed class Fixture:IDisposable
    {
        private static readonly string Parent=Path.GetFullPath(Path.Combine(AppContext.BaseDirectory,"hint-fixtures"));
        public string Root {get;}=Path.Combine(Parent,Guid.NewGuid().ToString("N"));public string Accounts=>Path.Combine(Root,"accounts.json");public string Security=>Path.Combine(Root,"security.golock");
        public Fixture()=>Directory.CreateDirectory(Root);
        public AppLockService Service(IAppLockStorage? storage=null)=>new(storage??new AppLockFileStorage(Security));
        public LocalStore Store(AppLockService service)=>new(false,Accounts,service);
        public AppLockService Seed(string hint,IAppLockStorage? storage=null){var service=Service(storage);Store(service).Save(State());Check(service.Enable(Password,out var error,hint),error);return service;}
        public void Dispose(){var full=Path.GetFullPath(Root);if(!full.StartsWith(Parent+Path.DirectorySeparatorChar,StringComparison.OrdinalIgnoreCase)||!Guid.TryParseExact(Path.GetFileName(full),"N",out _))throw new IOException("Unsafe hint fixture root");Directory.Delete(full,true);}
    }
}
